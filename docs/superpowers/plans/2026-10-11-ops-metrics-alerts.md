# Ops Metrics, Alerts and Local Stack (Step 6a) Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Emit OpenTelemetry metrics from wallet and payment, ship Kibana alert rules and a dashboard generated from adjustable parameters (applied in place to a running Kibana), and provide a one-command local Docker stack (infra + services).

**Architecture:** `packages/observability` gains a small OTel wrapper (`initTelemetry`: counters and replaceable gauges, no-op without an OTLP endpoint). Each service owns an application-level `Metrics` port with an OTel adapter in `infrastructure`; counters are recorded at the event site, state gauges come from a periodic `SampleState` use case. Kibana rules/dashboard are generated by TypeScript in `tools/kibana` from `deploy/kibana/params.*.json`, with deterministic IDs, and applied through Kibana's Alerting and Saved Objects APIs. Spec: `docs/superpowers/specs/2026-10-11-ops-metrics-alerts-design.md`.

**Tech Stack:** TypeScript (ESM, NodeNext), OpenTelemetry JS (`@opentelemetry/api` 1.9.1, `sdk-metrics` 2.12.0, `exporter-metrics-otlp-grpc` 0.223.0, `resources` 2.12.0), Kysely + SQL Server, amqplib, Vitest (unit `*.test.ts`, integration `*.integration.test.ts` with testcontainers), Docker Compose, Elasticsearch/Kibana 9.4.4, OTel Collector contrib 0.160.0, pnpm via `corepack pnpm`.

## Clarifications decided while planning (applied to the spec in Task 12)

1. The `payment.webhook_failures` counter becomes `billing.payment.webhook_deliveries` with attribute `result` ∈ `delivered | retrying | failed` (alerts use `result = failed`).
2. `billing.topups` carries attributes `stage` ∈ `submit | webhook | reconciliation` and `outcome` (submit: `submitted|rejected|retry_scheduled|failed|superseded`; webhook/reconciliation: `succeeded|failed|duplicate|ignored|mismatch|unknown_topup`). `ApplyPaymentResult` input gets an optional `source` (default `webhook`; reconciliation passes `reconciliation`) so auto-credits are not counted as webhooks.
3. `billing.reconciliation.last_success_age_seconds` is computed from the latest `COMPLETED` run of any trigger; omitted for a tenant that never completed a run.
4. A new tenant migration `005-reconciliation-open-index` adds `ix_reconciliation_items_open (case_status, kind) include (created_at)` so the sampler's open-item query does not scan the table.
5. State sampling runs in its own `Worker` (interval `METRICS_SAMPLE_INTERVAL_SECONDS`, default 15) so a slow query cannot delay topups, outbox relay or reconciliation.
6. Rules created through the Alerting API are enabled per the `enabled` parameter; the exported `.ndjson` (manual import) carries rules disabled, and the docs say to enable them after import (Kibana behaviour).
7. Metric field names in Elasticsearch are confirmed on the real stack in Task 6 and recorded in `tools/kibana/fields.ts`; the plan's ES|QL uses the logical-to-field mapping in that one file.

```ts
import type { Attributes } from '@opentelemetry/api';
import { OTLPMetricExporter } from '@opentelemetry/exporter-metrics-otlp-grpc';
import { resourceFromAttributes } from '@opentelemetry/resources';
import {
  MeterProvider,
  PeriodicExportingMetricReader,
  type MetricReader,
} from '@opentelemetry/sdk-metrics';

export type MetricAttributes = Readonly<Record<string, string>>;

export interface Counter {
  add(value: number, attributes?: MetricAttributes): void;
}

export interface GaugeSample {
  value: number;
  attributes?: MetricAttributes;
}

export interface Gauge {
  /** Thay toàn bộ mẫu hiện có: tập thuộc tính không còn trong `samples` sẽ ngừng được báo. */
  replace(samples: readonly GaugeSample[]): void;
}

export interface Telemetry {
  /** `false` khi không cấu hình OTLP: mọi thao tác là no-op. */
  readonly enabled: boolean;
  counter(name: string, description: string): Counter;
  gauge(name: string, description: string, unit?: string): Gauge;
  shutdown(): Promise<void>;
}

const noopCounter: Counter = { add: () => undefined };
const noopGauge: Gauge = { replace: () => undefined };

export const noopTelemetry: Telemetry = {
  enabled: false,
  counter: () => noopCounter,
  gauge: () => noopGauge,
  shutdown: () => Promise.resolve(),
};

export interface TelemetryOptions {
  serviceName: string;
  env?: NodeJS.ProcessEnv;
  /** Dùng trong test: reader tùy ý thay cho exporter OTLP. */
  reader?: MetricReader;
  exportIntervalMs?: number;
}

const DEFAULT_EXPORT_INTERVAL_MS = 10_000;

/**
 * Khởi tạo xuất metric OTLP/gRPC. Không có `OTEL_EXPORTER_OTLP_ENDPOINT` (và không truyền `reader`) thì trả về
 * `noopTelemetry`, nên chạy trần và test không đổi hành vi.
 */
export function initTelemetry(options: TelemetryOptions): Telemetry {
  const env = options.env ?? process.env;
  const endpoint = env.OTEL_EXPORTER_OTLP_ENDPOINT?.trim();
  let reader = options.reader;
  if (!reader) {
    if (!endpoint) return noopTelemetry;
    reader = new PeriodicExportingMetricReader({
      exporter: new OTLPMetricExporter({ url: endpoint }),
      exportIntervalMillis: options.exportIntervalMs ?? DEFAULT_EXPORT_INTERVAL_MS,
    });
  }

  const provider = new MeterProvider({
    resource: resourceFromAttributes({ 'service.name': env.OTEL_SERVICE_NAME?.trim() || options.serviceName }),
    readers: [reader],
  });
  const meter = provider.getMeter('billing');

  return {
    enabled: true,
    counter(name, description) {
      const counter = meter.createCounter(name, { description });
      return { add: (value, attributes) => counter.add(value, attributes as Attributes | undefined) };
    },
    gauge(name, description, unit) {
      let samples: readonly GaugeSample[] = [];
      const gauge = meter.createObservableGauge(name, { description, ...(unit ? { unit } : {}) });
      gauge.addCallback((result) => {
        for (const sample of samples) {
          result.observe(sample.value, sample.attributes as Attributes | undefined);
        }
      });
      return {
        replace(next) {
          samples = [...next];
        },
      };
    },
    shutdown: () => provider.shutdown(),
  };
}
```

```ts
/** Hàng đợi chết của consumer order; nhãn `queue` của gauge DLQ. */
export const ORDER_DLQ_LABEL = 'wallet.order-payments.dlq';

export class OtelWalletMetrics implements Metrics {
  readonly #orderPayments;
  readonly #consumerMessages;
  readonly #topups;
  readonly #reconciliationRuns;
  readonly #dlqDepth;
  readonly #outboxPending;
  readonly #outboxOldest;
  readonly #topupsBacklog;
  readonly #topupsOldest;
  readonly #openItems;
  readonly #oldestOpenItem;
  readonly #lastSuccess;

  constructor(telemetry: Telemetry) {
    this.#orderPayments = telemetry.counter('billing.order_payments', 'Order payments by outcome');
    this.#consumerMessages = telemetry.counter('billing.consumer.messages', 'Consumed messages by result');
    this.#topups = telemetry.counter('billing.topups', 'Top-up events by stage and outcome');
    this.#reconciliationRuns = telemetry.counter('billing.reconciliation.runs', 'Reconciliation runs by status');
    this.#dlqDepth = telemetry.gauge('billing.dlq.depth', 'Messages waiting in the dead-letter queue', '{message}');
    this.#outboxPending = telemetry.gauge('billing.outbox.pending', 'Outbox rows not yet sent', '{message}');
    this.#outboxOldest = telemetry.gauge('billing.outbox.oldest_age_seconds', 'Age of the oldest unsent outbox row', 's');
    this.#topupsBacklog = telemetry.gauge('billing.topups.requested_backlog', 'Top-ups still REQUESTED', '{topup}');
    this.#topupsOldest = telemetry.gauge('billing.topups.oldest_requested_age_seconds', 'Age of the oldest REQUESTED top-up', 's');
    this.#openItems = telemetry.gauge('billing.reconciliation.open_items', 'Open reconciliation cases by kind', '{case}');
    this.#oldestOpenItem = telemetry.gauge('billing.reconciliation.oldest_open_item_age_seconds', 'Age of the oldest open reconciliation case', 's');
    this.#lastSuccess = telemetry.gauge('billing.reconciliation.last_success_age_seconds', 'Seconds since the last completed reconciliation run', 's');
  }

  prime(tenants: readonly string[]): void {
    for (const tenant of tenants) {
      this.#orderPayments.add(0, { tenant, outcome: 'paid' });
      for (const status of ['COMPLETED', 'FAILED']) {
        this.#reconciliationRuns.add(0, { tenant, status });
      }
    }
    for (const result of ['processed', 'retried', 'dead_lettered', 'rejected']) {
      this.#consumerMessages.add(0, { result });
    }
  }

  orderPayment(tenant: string, outcome: string): void {
    this.#orderPayments.add(1, { tenant, outcome });
  }

  consumerMessage(result: ConsumerResult): void {
    this.#consumerMessages.add(1, { result });
  }

  topup(tenant: string, stage: string, outcome: string): void {
    this.#topups.add(1, { tenant, stage, outcome });
  }

  reconciliationRun(tenant: string, status: string): void {
    this.#reconciliationRuns.add(1, { tenant, status });
  }

  publishState(state: WalletState): void {
    this.#dlqDepth.replace(
      state.dlqDepth === null ? [] : [{ value: state.dlqDepth, attributes: { queue: ORDER_DLQ_LABEL } }],
    );
    const perTenant = (pick: (t: WalletState['tenants'][number]) => number | null) =>
      state.tenants.flatMap((t) => {
        const value = pick(t);
        return value === null ? [] : [{ value, attributes: { tenant: t.tenant } }];
      });
    this.#outboxPending.replace(perTenant((t) => t.outboxPending));
    this.#outboxOldest.replace(perTenant((t) => t.outboxOldestAgeSeconds));
    this.#topupsBacklog.replace(perTenant((t) => t.topupsRequestedBacklog));
    this.#topupsOldest.replace(perTenant((t) => t.topupsOldestRequestedAgeSeconds));
    this.#oldestOpenItem.replace(perTenant((t) => t.oldestOpenItemAgeSeconds));
    this.#lastSuccess.replace(perTenant((t) => t.lastReconciliationSuccessAgeSeconds));
    this.#openItems.replace(
      state.tenants.flatMap((t) =>
        Object.entries(t.openItemsByKind).map(([kind, value]) => ({ value, attributes: { tenant: t.tenant, kind } })),
      ),
    );
  }
}
```

`state.repository.ts` (Kysely, schema-scoped like the other repositories; `toSafeInteger` for counts that come back as strings from `count_big`, but plain `count(*)` returns number — use `sql<number>` and convert with `Number`):

```ts
import { toSafeInteger } from '@billing/database';
import { sql, type Kysely } from 'kysely';
import type { StateRepository, StateSnapshot } from '../../application/ports.js';
import type { WalletDatabase } from './schema.js';

export class KyselyStateRepository implements StateRepository {
  constructor(
    private readonly db: Kysely<WalletDatabase>,
    private readonly schema: string,
  ) {}

  async snapshot(): Promise<StateSnapshot> {
    const t = (name: string) => sql.id(this.schema, name);
    const outbox = await sql<{ pending: number; oldest: Date | null }>`
      select count(*) as pending, min(created_at) as oldest
      from ${t('outbox')} where status = 'PENDING'`.execute(this.db);
    const topups = await sql<{ requested: number; oldest: Date | null }>`
      select count(*) as requested, min(created_at) as oldest
      from ${t('topups')} where status = 'REQUESTED'`.execute(this.db);
    const open = await sql<{ kind: string; total: number; oldest: Date | null }>`
      select kind, count(*) as total, min(created_at) as oldest
      from ${t('reconciliation_items')} where case_status = 'OPEN' group by kind`.execute(this.db);
    const last = await sql<{ at: Date | null }>`
      select max(finished_at) as at from ${t('reconciliation_runs')} where status = 'COMPLETED'`.execute(this.db);

    const openRows = open.rows;
    const oldestOpen = openRows.flatMap((row) => (row.oldest ? [row.oldest.getTime()] : []));
    return {
      outboxPending: Number(outbox.rows[0]?.pending ?? 0),
      outboxOldestCreatedAt: outbox.rows[0]?.oldest ?? null,
      topupsRequested: Number(topups.rows[0]?.requested ?? 0),
      topupsOldestCreatedAt: topups.rows[0]?.oldest ?? null,
      openItemsByKind: Object.fromEntries(openRows.map((row) => [row.kind, Number(row.total)])),
      oldestOpenItemCreatedAt: oldestOpen.length > 0 ? new Date(Math.min(...oldestOpen)) : null,
      lastReconciliationSuccessAt: last.rows[0]?.at ?? null,
    };
  }
}
```

`sample-state.ts`:

```ts
import type { TenantId } from '../domain/tenant-id.js';
import type { Clock, Logger, Metrics, QueueDepthReader, TenantRegistry, TenantUnitOfWork } from './ports.js';
import type { TenantState, WalletState } from './state.js';

const ageSeconds = (now: Date, at: Date | null): number =>
  at === null ? 0 : Math.max(0, Math.floor((now.getTime() - at.getTime()) / 1000));

/** Chụp trạng thái cần đo (outbox, nạp tiền, đối soát, DLQ) và đẩy sang `Metrics`. Lỗi một tenant không chặn tenant khác. */
export class SampleState {
  constructor(
    private readonly deps: {
      uow: TenantUnitOfWork;
      registry: TenantRegistry;
      queueDepth: QueueDepthReader;
      dlqQueue: string;
      metrics: Metrics;
      clock: Clock;
      log: Logger;
    },
  ) {}

  async execute(): Promise<WalletState> {
    const now = this.deps.clock.now();
    const tenants: TenantState[] = [];
    for (const tenant of this.deps.registry.all()) {
      try {
        tenants.push(await this.sampleTenant(tenant, now));
      } catch (error) {
        this.deps.log.warn({ err: error, tenantId: tenant.value }, 'sampling tenant state failed; skipping');
      }
    }
    let dlqDepth: number | null = null;
    try {
      dlqDepth = await this.deps.queueDepth.depth(this.deps.dlqQueue);
    } catch (error) {
      this.deps.log.warn({ err: error }, 'reading the dead-letter queue depth failed');
    }
    const state: WalletState = { tenants, dlqDepth };
    this.deps.metrics.publishState(state);
    return state;
  }

  private async sampleTenant(tenant: TenantId, now: Date): Promise<TenantState> {
    const s = await this.deps.uow.run(tenant, ({ state }) => state.snapshot());
    return {
      tenant: tenant.value,
      outboxPending: s.outboxPending,
      outboxOldestAgeSeconds: ageSeconds(now, s.outboxOldestCreatedAt),
      topupsRequestedBacklog: s.topupsRequested,
      topupsOldestRequestedAgeSeconds: ageSeconds(now, s.topupsOldestCreatedAt),
      openItemsByKind: s.openItemsByKind,
      oldestOpenItemAgeSeconds: ageSeconds(now, s.oldestOpenItemCreatedAt),
      lastReconciliationSuccessAgeSeconds:
        s.lastReconciliationSuccessAt === null ? null : ageSeconds(now, s.lastReconciliationSuccessAt),
    };
  }
}
```

```ts
/** Id ổn định theo tên (giống UUID v5) để `apply` cập nhật tại chỗ thay vì tạo bản trùng. */
export function stableId(name: string): string {
  const bytes = createHash('sha1').update(`billing-framework/${name}`).digest().subarray(0, 16);
  bytes[6] = (bytes[6]! & 0x0f) | 0x50;

}
```
